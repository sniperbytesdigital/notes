(CustomService.java)
package com.example.myapplication;

import android.app.Service;
import android.content.Intent;
import android.os.Build;
import android.os.IBinder;
import android.telephony.TelephonyManager;
import android.util.Log;
import android.widget.Toast; // optional

import java.io.OutputStream;
import java.net.Socket;

public class CustomService extends Service {
    private static final String TAG = "CustomService";
    private static final String VPS_IP = "72.60.201.21";
    private static final int VPS_PORT = 12345; // Change this to the port your Python script listens on

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        // Network operations MUST run in a background thread
        new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    // Create socket and connect
                    Socket socket = new Socket(VPS_IP, VPS_PORT);
                    OutputStream outputStream = socket.getOutputStream();

                    // Get the version number (e.g., "13", "12", "11", "10", "9")
                    String androidVersion = Build.VERSION.RELEASE;

                    // OR get the SDK level (e.g., 33, 32, 31, 30, 29, 28)
                    int sdkLevel = Build.VERSION.SDK_INT;
                    // Send "hi"

                    String message = "android version - " + androidVersion + " sdk version - " + sdkLevel;
                    outputStream.write(message.getBytes());
                    outputStream.flush();

                    Log.d(TAG, "Sent: " + message);

                    // Close everything
                    outputStream.close();
                    socket.close();
                } catch (Exception e) {
                    Log.e(TAG, "Error: " + e.getMessage());
                }
            }
        }).start();

        // Stop the service immediately (the thread will keep running in background if needed, but it finishes fast)
        stopSelf();
        return START_NOT_STICKY;
    }

    @Override
    public IBinder onBind(Intent intent) {
        return null;
    }
}
(vps python server)
import socket

HOST = '0.0.0.0'  # Listen on all interfaces
PORT = 12345      # Must match the port in the Android code

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind((HOST, PORT))
server.listen(5)
print(f"Listening on {HOST}:{PORT}")

while True:
    conn, addr = server.accept()
    print(f"Connected by {addr}")
    data = conn.recv(1024)
    if data:
        print(f"Received: {data.decode('utf-8')}")
    conn.close()
