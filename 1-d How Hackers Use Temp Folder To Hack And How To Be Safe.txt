import zipfile, os
from pathlib import Path

desktop = Path.home() / "Desktop"
backup = Path(os.environ['TEMP']) / "desktop_backup.zip"

with zipfile.ZipFile(backup, 'w') as z:
    for file in os.listdir(str(desktop)):
        file_path = desktop / file
        if os.path.isfile(file_path):
            z.write(file_path, file)

print(f"Backup created: {backup}")
